## Wind
wind_pressure_mm = 79                          # V3: "The organ was voiced on 79 mm of wind pressure"
bellows = 4 wedge-shaped single-fold, 4' x 8', weighted with lead shot, foot-pumpable, electric blower   # V3
bellows_bloom = as the wedge bellows close, air velocity and pressure rise slightly ("crescendo or bloom")  # V3 footnote
offset_32 = the 12 lowest notes of Subbass 32' and Trombone 32' stand on electric offset chests fed directly from the blower   # V3

## Compass
manual_compass = C-a3, 58 notes               # P: "Manual I/II/III, 58 notes (C-a3)"
pedal_compass  = C-f1, 30 notes, flat/parallel   # P
meantone_manual_compass = 48 notes: C, D, E-c3, d3   # P (no C#, D#, c#3; nothing above d3)
meantone_pedal_compass  = 28 notes: C, D, E-f1       # P

## Names. The scale sheets and the stop list name three stops differently:
##   OW "Praestant 8'" (sheets) = "Principal 8'" (stop list)
##   Ped "Praestant 16'" (sheets) = "Principal 16'" (stop list)
##   Ped "Gedeckt 8'" (sheets) = "Bourdon 8'" (stop list)
## Transmissions. V1 marks Ped Subbass 16', Octave 8' and Posaune 16' "octave transmission"; which
## rank each is drawn from is not stated. What the scales show:
##   Subbass 32' c0 (194x228, cut-up 84) = Subbass 16' C (194x228): the 32' probably has 12 pipes of
##     its own (the offset-chest basses, V3) and continues on the 16' rank.
##   HW Praestant 16' C-C# are the wood Haskell pipes of the Pedal Praestant 16' (flue sheet note).
##   Trombone 32' c0 (poplar 194x194, 4480) and Posaune 16' C (metal 220, 4480) differ in material:
##     these are separate pipes.
##   Pedal Octave 8' C (155) is not HW Praestant 16' c0 (150): separate pipes, sheet C-c2.
